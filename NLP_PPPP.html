# BUILD OUR FIRST CLEANING FUUNCTION

import re

def clean_text(text):
    text = str(text)
    
    # Convert to lowercase
    text = text.lower()
    
    # Replace placeholders
    for old, new in placeholder_map.items():
        text = text.replace(old.lower(), new.lower())
    
    # Remove special characters
    text = re.sub(r"[^a-zA-Z0-9_\s]", " ", text)
    
    # Remove extra spaces
    text = re.sub(r"\s+", " ", text).strip()
    
    return text

# TICKET CLASSIFICATION

def classify_ticket(ticket_text, tfidf_vectorizer, trained_model):
    """
    Takes a raw customer support ticket string and returns the predicted intent 
    and model confidence percentage.
    """
    # 1. Vectorize the input text using the trained TF-IDF vectorizer
    input_vector = tfidf_vectorizer.transform([ticket_text])
    
    # 2. Predict the intent category
    predicted_intent = trained_model.predict(input_vector)[0]
    
    # 3. Calculate prediction confidence (probability score)
    probabilities = trained_model.predict_proba(input_vector)[0]
    confidence_score = np.max(probabilities) * 100
    
    return predicted_intent, confidence_score

# --- Example Usage ---

# AUTOMATIC ROUTING


# 1. Define routing destinations based on intent categories
AUTOMATIC_ROUTING_TABLE = {
    # Self-Service Automation
    "recover_password": {"route": "Automated Auth Bot", "type": "Self-Service API"},
    "track_order": {"route": "Logistics API", "type": "Self-Service API"},
    "track_refund": {"route": "Finance Portal Bot", "type": "Self-Service API"},
    
    # Specialized Department Queues
    "payment_issue": {"route": "Billing Escalations Team", "type": "Human Queue"},
    "complaint": {"route": "Customer Care Supervisor", "type": "Human Queue"},
    "delete_account": {"route": "Security & Privacy Team", "type": "Human Queue"}
}

def auto_route_ticket(ticket_text, tfidf_vectorizer, trained_model, confidence_threshold=0.70):
    """
    Classifies a raw ticket and automatically routes it based on confidence score.
    """
    # Step A: Transform raw text and get model prediction probabilities
    vec = tfidf_vectorizer.transform([ticket_text])
    probabilities = trained_model.predict_proba(vec)[0]
    
    # Step B: Get predicted intent and maximum confidence score
    top_class_idx = probabilities.argmax()
    predicted_intent = trained_model.classes_[top_class_idx]
    confidence = probabilities[top_class_idx]
    
    # Step C: Routing Guardrail Logic
    if confidence < confidence_threshold:
        return {
            "ticket": ticket_text,
            "predicted_intent": predicted_intent,
            "confidence": f"{confidence * 100:.2f}%",
            "destination": "Human Triage & Manual Review",
            "routing_type": "Manual Escalation (Low Confidence)",
            "status": "Flagged"
        }
    
    # Step D: Match with Routing Table or Default Queue
    route_data = AUTOMATIC_ROUTING_TABLE.get(predicted_intent, {
        "route": "General Tier-1 Support Queue",
        "type": "Standard Human Assignment"
    })
    
    return {
        "ticket": ticket_text,
        "predicted_intent": predicted_intent,
        "confidence": f"{confidence * 100:.2f}%",
        "destination": route_data["route"],
        "routing_type": route_data["type"],
        "status": "Auto-Routed"
    }

# --- Example Usage ---
sample_ticket = "Can you help me reset my password for my account?"
result = auto_route_ticket(sample_ticket, tfidf, model)

print("Ticket Result:")
for key, value in result.items():
    print(f"  {key}: {value}")

import streamlit as st
import numpy as np
import joblib

# Set up page configuration
st.set_page_config(page_title="NLP Ticket Routing System", page_icon="🎫", layout="centered")

# 1. Load saved model and vectorizer
@st.cache_resource
def load_assets():
    # Make sure you saved your trained model and tfidf objects using joblib
    tfidf = joblib.load("tfidf_vectorizer.pkl")
    model = joblib.load("logistic_regression_model.pkl")
    return tfidf, model

tfidf, model = load_assets()

# 2. Define Routing Table
AUTOMATIC_ROUTING_TABLE = {
    "recover_password": {"route": "Automated Auth Bot", "type": "Self-Service API"},
    "track_order": {"route": "Logistics API", "type": "Self-Service API"},
    "track_refund": {"route": "Finance Portal Bot", "type": "Self-Service API"},
    "payment_issue": {"route": "Billing Escalations Team", "type": "Human Queue"},
    "complaint": {"route": "Customer Care Supervisor", "type": "Human Queue"},
    "delete_account": {"route": "Security & Privacy Team", "type": "Human Queue"}
}

# 3. Build UI Layout
st.title("🎫 Customer Support Ticket Classifier & Auto-Router")
st.write("Enter a customer message below to automatically classify its intent and route it to the correct department.")

# User input text box
user_ticket = st.text_area("Customer Ticket Message:", placeholder="e.g., I was charged twice for my subscription last week.")

# Classify button
if st.button("Classify & Route Ticket"):
    if not user_ticket.strip():
        st.warning("Please enter a valid ticket message.")
    else:
        # Step A: Transform input text
        input_vector = tfidf.transform([user_ticket])
        
        # Step B: Get prediction and confidence
        probabilities = model.predict_proba(input_vector)[0]
        top_idx = probabilities.argmax()
        predicted_intent = model.classes_[top_idx]
        confidence = probabilities[top_idx] * 100
        
        # Step C: Routing Guardrail Logic
        confidence_threshold = 70.0
        if confidence < confidence_threshold:
            destination = "Human Triage & Manual Review Queue"
            routing_type = "Manual Review (Low Confidence)"
            status = "⚠️ Flagged for Triage"
        else:
            route_info = AUTOMATIC_ROUTING_TABLE.get(predicted_intent, {
                "route": "General Tier-1 Support Queue",
                "type": "Standard Human Assignment"
            })
            destination = route_info["route"]
            routing_type = route_info["type"]
            status = "✅ Auto-Routed"

        # Step D: Display Results
        st.divider()
        st.subheader("Classification & Routing Results")
        
        col1, col2 = st.columns(2)
        with col1:
            st.metric(label="Predicted Intent", value=predicted_intent)
            st.metric(label="Confidence Score", value=f"{confidence:.2f}%")
        
        with col2:
            st.metric(label="Routing Destination", value=destination)
            st.metric(label="Workflow Type", value=routing_type)
            
        if confidence < confidence_threshold:
            st.error(f"Status: {status} (Confidence below {confidence_threshold}%)")
        else:
            st.success(f"Status: {status}")
